# ===================================================================
#  QUALYS EVIDENCE - BULK READ-ONLY COLLECTION (all 7 controls)
#  Run from a PowerCLI window ALREADY connected to vCenter:
#      C:\Users\Samoal\Desktop\Get-QualysEvidence.ps1
#  Targets: C:\temp\vmlist.txt  (one vCenter VM inventory name per line)
#  Read-only: no config changes, no reboots. Only guest disk activity
#  is a GUID-named temp file from "secedit /export", deleted in-script.
# ===================================================================

if (-not ($global:DefaultVIServers | Where-Object { $_.IsConnected })) {
    throw "Not connected to vCenter. Run  Connect-VIServer <vcenter>  first, then re-run this."
}

$listPath = 'C:\temp\vmlist.txt'
if (-not (Test-Path $listPath)) { throw "VM list not found: $listPath" }
$VMs = Get-Content $listPath | ForEach-Object { $_.Trim() } | Where-Object { $_ -and $_ -notmatch '^#' }
if (-not $VMs) { throw "No VM names found in $listPath" }

$GuestCred = Get-Credential -Message 'Guest LOCAL ADMIN credential for the target VMs (e.g. .\Administrator)'

$OutDir = "C:\temp\qualys_evidence\$(Get-Date -Format yyyyMMdd_HHmmss)"
New-Item -ItemType Directory -Path $OutDir -Force | Out-Null

$GuestScript = @'
$ErrorActionPreference = "Continue"
"HOST         : $(hostname)"
"COMPUTERNAME : $env:COMPUTERNAME"
$os = Get-CimInstance Win32_OperatingSystem
"OS           : $($os.Caption) | Build $($os.BuildNumber) | $($os.Version)"
$dg = "HKLM:\SOFTWARE\Policies\Microsoft\Windows\DeviceGuard"

""
"===== 25349  Kernel-mode Hardware-enforced Stack Protection ====="
"ConfigureKernelShadowStacksLaunch = " + (Get-ItemProperty $dg -Name ConfigureKernelShadowStacksLaunch -EA SilentlyContinue).ConfigureKernelShadowStacksLaunch
Get-CimInstance -Namespace root\Microsoft\Windows\DeviceGuard -ClassName Win32_DeviceGuard |
    Format-List AvailableSecurityProperties, SecurityServicesConfigured, SecurityServicesRunning, VirtualizationBasedSecurityStatus, CodeIntegrityPolicyEnforcementStatus

""
"===== 4501  Audit Policy Change ====="
auditpol /get /subcategory:"Audit Policy Change"

""
"===== 26502  Allow Administrator Account Lockout ====="
net accounts
$t2 = Join-Path $env:TEMP ("q_secpol_{0}.cfg" -f [guid]::NewGuid())
secedit /export /cfg $t2 /areas SECURITYPOLICY | Out-Null
Get-Content $t2 | Select-String "AllowAdministratorLockout|LockoutBadCount|LockoutDuration|ResetLockoutCount" | ForEach-Object { $_.Line.Trim() }
Remove-Item $t2 -Force -EA SilentlyContinue

""
"===== 2200 / 2196  Deny RDP logon / Deny network logon ====="
$t1 = Join-Path $env:TEMP ("q_secpol_{0}.cfg" -f [guid]::NewGuid())
secedit /export /cfg $t1 /areas USER_RIGHTS | Out-Null
Get-Content $t1 | Select-String "SeDenyRemoteInteractiveLogonRight|SeDenyNetworkLogonRight" | ForEach-Object { $_.Line.Trim() }
Remove-Item $t1 -Force -EA SilentlyContinue

""
"===== 10473  Credential Guard (LsaCfgFlags) ====="
"LsaCfgFlags = " + (Get-ItemProperty $dg -Name LsaCfgFlags -EA SilentlyContinue).LsaCfgFlags

""
"===== 10474  HVCI (HypervisorEnforcedCodeIntegrity) ====="
"HypervisorEnforcedCodeIntegrity = " + (Get-ItemProperty $dg -Name HypervisorEnforcedCodeIntegrity -EA SilentlyContinue).HypervisorEnforcedCodeIntegrity
'@

foreach ($name in $VMs) {
    $file = Join-Path $OutDir ("{0}_qualys_evidence.txt" -f $name)
    "===== VM(vCenter): $name | $(Get-Date -Format s) =====" | Set-Content $file

    $vm = Get-VM -Name $name -ErrorAction SilentlyContinue
    if     (-not $vm)                       { "SKIPPED: not in vCenter inventory"          | Add-Content $file }
    elseif ($vm.PowerState -ne 'PoweredOn') { "SKIPPED: power state = $($vm.PowerState)"    | Add-Content $file }
    elseif ($vm.Guest.State -ne 'Running')  { "SKIPPED: VMware Tools = $($vm.Guest.State)"  | Add-Content $file }
    else {
        try {
            $p = @{ VM = $vm; GuestCredential = $GuestCred; ScriptType = 'Powershell'; ScriptText = $GuestScript; ErrorAction = 'Stop' }
            (Invoke-VMScript @p).ScriptOutput.TrimEnd() | Add-Content $file
        } catch {
            "ERROR: $($_.Exception.Message)" | Add-Content $file
        }
    }

    Write-Host ""
    Write-Host ("#" * 72) -ForegroundColor Cyan
    Get-Content $file | Write-Host
}

Write-Host ""
Write-Host ("=" * 72) -ForegroundColor Green
Write-Host "DONE - $($VMs.Count) VM(s). Evidence saved to: $OutDir" -ForegroundColor Green
Get-ChildItem $OutDir | Select-Object Name, Length, LastWriteTime | Format-Table -AutoSize
explorer $OutDir
